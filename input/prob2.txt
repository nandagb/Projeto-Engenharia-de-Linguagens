vazio func main() {
   int c1 = 0;
   int c2 = 0;
   int c3 = 0;
   int c4 = 0;
   
   int num = 0;

   entrada(num);

   enquanto (num >= 0) {
      
      # Verifica os intervalos
      se (num <= 25) {
         c1++;
      } 
      senao se (num <= 50) {
         c2++;
      } 
      senao se (num <= 75) {
         c3++;
      } 
      senao se (num <= 100) {
         c4++;
      }

      entrada(num);
   }

   saida(c1);
   saida(c2);
   saida(c3);
   saida(c4);
}
