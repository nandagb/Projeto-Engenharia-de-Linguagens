vazio func main() {
   int linhas1;
   int colunas1;
   int linhas2;
   int colunas2;

   #Leitura das dimensões
   saida("Digite a largura da primeira matriz: \n");
   entrada(linhas1);
   saida("Digite a altura da primeira matriz: \n");
   entrada(colunas1);
   saida("Digite a largura da segunda matriz: \n");
   entrada(linhas2);
   saida("Digite a altura da segunda matriz: \n");
   entrada(colunas2);

   Lista<Lista<int>> matriz1 = novo Lista<>()(linhas1);
   Lista<Lista<int>> matriz2 = novo Lista<>()(linhas2);

    repita (int i = 0, i < linhas1, i++) {
        Lista<int> coluna1 = novo Lista<>()(colunas1);
        repita (int j=0, j<colunas1, j++) {
            int k;
            saida("Digite o valor da posição (" + texto(i) + ", " + texto(j) + ") da matriz 1: \n");
            entrada(k);
            coluna1[j] = k;
        }

      matriz1[i] = coluna1;
    }

    repita (int i = 0, i < linhas2, i++) {
        Lista<int> coluna2 = novo Lista<>()(colunas2);
        repita (int j=0, j<colunas2, j++) {
            int x;
            saida("Digite o valor da posição (" + texto(i) + ", " + texto(j) + ") da matriz 2: \n");
            entrada(x);
            coluna2[j] = x;
        }

      matriz2[i] = coluna2;
    }

    saida("Matriz1: " + "\n");
    repita (int i = 0, i < linhas1, i++) {
        repita (int j=0, j < colunas1, j++) {
            saida(texto(matriz1[i][j]) + " ");
        }
        saida("\n");
    }

    saida("Matriz2: " + "\n");
    repita (int i = 0, i < linhas2, i++) {
        repita (int j=0, j < colunas2, j++) {
            saida(texto(matriz2[i][j]) + " ");
        }
        saida("\n");
    }

    se (linhas1 == linhas2 && colunas1 == colunas2) {
        Lista<Lista<int>> soma = novo Lista<>()(linhas1);

        saida("Somando matrizes...\n");
        repita (int i = 0, i < linhas1, i++) {
            Lista<int> coluna_soma = novo Lista<>()(colunas1);
            repita (int j = 0, j < colunas1, j++) {
                int result_soma = matriz1[i][j] + matriz2[i][j];
                coluna_soma[j] = result_soma;
                saida(texto(result_soma) + " ");
            }
            printf("\n");
            soma[i] = coluna_soma;
        }
    }
    senao {
        saida("Não foi possível somar as matrizes!\n");
    }

    se (colunas1 == linhas2) {
        saida("Multipliando as matrizes...\n");
        Lista<Lista<int>> prod = novo Lista<>()(linhas1);
        # float prod[l1][c2];

        # Inicializa resultado
        repita (int i = 0, i < linhas1, i++){
            Lista<int> linha_prod = novo Lista<>()(colunas2);
            repita (int j = 0, j < colunas2, j++){
                linha_prod[j] = 0;
            }
            prod[i] = linha_prod;
        }

        # Multiplicação
        repita (int i = 0, i < linhas1, i++) {
            repita (int j = 0, j < colunas2, j++) {
                repita (int e = 0, e < colunas1, e++) {
                    prod[i][j] = prod[i][j] + matriz1[i][e] * matriz2[e][j];
                }
            }
        }

        saida("\nProduto das matrizes:\n");
        repita (int i = 0, i < linhas1, i++) {
            repita (int j = 0, j < colunas2, j++) {
                saida(texto(prod[i][j]) + " ");
            }
            saida("\n");
        }
    }
    senao {
        saida("Não foi possível fazer o produto das matrizes!\n");
    }


}
