vazio func main() {

   entrada(linhas2);
   saida("Digite a altura da segunda matriz: \n");

   Lista<Lista<int>> matriz2 = novo Lista<>()(colunas2);

    repita (int i = 0, i < colunas1, i++) {
        Lista<int> linha1 = novo Lista<>()(linhas1);
        repita (int j=0, j<linhas1, j++) {

        }

      matriz1[i] = linha1;
    }

    repita (int n = 0, n < colunas2, n++) {
        Lista<int> linha2 = novo Lista<>()(linhas2);

            int x;
            saida("Digite o valor da posição (" + texto(n) + ", " + texto(o) + ") da matriz 2: \n");
            entrada(x);
            linha2[o] = x;
        }

      matriz2[n] = linha2;
    }

    saida("Matriz1: " + "\n");
    repita (int l = 0, l < colunas1, l++) {
        repita (int m=0, m < linhas1, m++) {
            saida(texto(matriz1[l][m]) + " ");
        }
        saida("\n");
    }

    saida("Matriz2: " + "\n");
    repita (int y = 0, y < colunas2, y++) {
        repita (int z=0, z < linhas2, z++) {
            saida(texto(matriz2[y][z]) + " ");
        }
        saida("\n");
    }

    se (linhas1 == linhas2 && colunas1 == colunas2) {
        Lista<Lista<int>> soma = novo Lista<>()(colunas1);

        saida("Somando matrizes...\n");
        repita (int ic = 0, ic < colunas1, ic++) {
            Lista<int> linha_soma = novo Lista<>()(linhas1);
            repita (int jl = 0, jl < linhas1, jl++) {
                int result_soma = matriz1[ic][jl] + matriz2[ic][jl];
                linha_soma[jl] = result_soma;
                saida(texto(result_soma) + " ");
            }
            printf("\n");
            soma[n] = linha_soma;
